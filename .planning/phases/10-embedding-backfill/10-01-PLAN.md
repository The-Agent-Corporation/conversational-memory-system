implementations, so the Phase 10 backfill command can commit ~500 vectors per SQLite transaction instead

```typescript
// saveSummaryVector: validates dimensions === vector.length, validates every value
//   Number.isFinite(Math.fround(value)), then
//   `insert into summary_vectors (...) values (...) on conflict(summary_id, provider, model) do update set ...`
//
// deleteSummaryVectors — the transaction idiom to mirror:
async deleteSummaryVectors(summaryIds: string[]): Promise<void> {
  if (summaryIds.length === 0) return;
  this.database.exec("begin immediate");
  try {
    const statement = this.database.prepare("delete from summary_vectors where summary_id = ?");
    for (const summaryId of summaryIds) statement.run(summaryId);
    this.database.exec("commit");
  } catch (error) {
    this.database.exec("rollback");
    throw error;
  }
}
```
